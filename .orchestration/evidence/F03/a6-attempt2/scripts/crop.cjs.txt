// usage: node crop.cjs in.png out.png y0 h [scale]
const {PNG}=require(process.env.WT+'/node_modules/pngjs');
const fs=require('fs');
const [,,inp,out,y0s,hs,ss]=process.argv;
const src=PNG.sync.read(fs.readFileSync(inp));
const y0=+y0s, h=Math.min(+hs, src.height-y0), s=+(ss||1);
const W=Math.floor(src.width/s), H=Math.floor(h/s);
const dst=new PNG({width:W,height:H});
for(let y=0;y<H;y++)for(let x=0;x<W;x++){
  const si=((y0+y*s)*src.width+x*s)*4, di=(y*W+x)*4;
  for(let c=0;c<4;c++)dst.data[di+c]=src.data[si+c];
}
fs.writeFileSync(out,PNG.sync.write(dst));
console.log(out,W,H);
